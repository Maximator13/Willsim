import numpy as np


def moyenne_glissante(x, n=5):
    """Lisse le signal x par une moyenne glissante sur n échantillons."""
    noyau = np.ones(n) / n
    return np.convolve(x, noyau, mode="valid")  # len(x) - n + 1 valeurs
